\begin{tikzpicture}[
  x=1cm,
  y=1cm,
  line cap=round,
  line join=round,
  every node/.style={font=\small}
]
  % 以悬点为圆心，O 和 p 同在半径为 ell 的圆弧上。
  \def\radius{3}
  \def\swingangle{35}
  \def\gravitylength{2.0}
  \pgfmathsetmacro{\px}{\radius*sin(\swingangle)}
  \pgfmathsetmacro{\py}{\radius-\radius*cos(\swingangle)}
  \pgfmathsetmacro{\topEndAngle}{270+\swingangle}
  \pgfmathsetmacro{\pEndAngle}{270+\swingangle}
  \pgfmathsetmacro{\pLabelAngle}{270+\swingangle/2}

  \coordinate (C) at (0,\radius);
  \coordinate (O) at (0,0);
  \coordinate (P) at (\px,\py);
  \coordinate (GravityEnd) at (\px,{\py-\gravitylength});
  \coordinate (Kleft) at ({\radius*cos(200)},{\radius+\radius*sin(200)});
  \coordinate (Kright) at ({\radius*cos(338)},{\radius+\radius*sin(338)});
  \coordinate (TopAngleStart) at ({0.50*cos(270)},{\radius+0.50*sin(270)});
  \coordinate (TopAngleLabel) at ({0.82*cos(270+\swingangle/2)},{\radius+0.82*sin(270+\swingangle/2)});
  \coordinate (PAngleStart) at ({\px+0.48*cos(270)},{\py+0.48*sin(270)});
  \coordinate (PAngleLabel) at ({\px+0.74*cos(\pLabelAngle)},{\py+0.74*sin(\pLabelAngle)});

  % 轨道圆弧，以原图的悬点为圆心。
  \draw[densely dashed, line width=0.7pt]
    (Kleft) arc[start angle=200,end angle=338,radius=\radius cm];

  % 悬点到最低点和质点的两条半径。
  \draw[line width=0.8pt] (C)--(O);
  \draw[line width=0.8pt] (C)--(P)
    node[pos=0.56,above right=2pt] {$\ell$};
  \node[right=2pt] at (0,1.52) {$\ell$};

  % p 点处按原图作完整的重力平行四边形分解。
  \pgfmathsetmacro{\componentEndX}
    {\px-\gravitylength*sin(\swingangle)*cos(\swingangle)}
  \pgfmathsetmacro{\componentEndY}
    {\py-\gravitylength*sin(\swingangle)*sin(\swingangle)}
  \pgfmathsetmacro{\radialEndX}
    {\px+\gravitylength*cos(\swingangle)*sin(\swingangle)}
  \pgfmathsetmacro{\radialEndY}
    {\py-\gravitylength*cos(\swingangle)*cos(\swingangle)}
  \draw[densely dashed, line width=0.7pt]
    (\componentEndX,\componentEndY)--(GravityEnd)
    --(\radialEndX,\radialEndY);
  \draw[->,line width=0.8pt] (P)--(GravityEnd)
    node[pos=0.62,left=3pt] {$mg$};
  \draw[->,line width=0.8pt] (P)--(\componentEndX,\componentEndY);
  \draw[->,line width=0.8pt] (P)--(\radialEndX,\radialEndY);

  % 两处角标相同：摆杆与竖直方向的夹角。
  \draw[line width=0.65pt] (TopAngleStart)
    arc[start angle=270,end angle=\topEndAngle,radius=0.50cm];
  \node at (TopAngleLabel) {$\theta$};
  \draw[line width=0.65pt] (PAngleStart)
    arc[start angle=270,end angle=\pEndAngle,radius=0.48cm];
  \node at (PAngleLabel) {$\theta$};

  \fill (C) circle (1.7pt);
  \fill (O) circle (1.7pt);
  \fill[white] (P) circle (1.8pt);
  \draw (P) circle (1.8pt);
  \node[below left=2pt] at (O) {$O$};
  \node[above=2pt] at (P) {$p$};
  \node[above right=1pt] at (Kright) {$K$};
\end{tikzpicture}
