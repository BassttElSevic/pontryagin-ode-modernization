  % A smooth illustrative ODE, not an additional example in the text:
  %   phi(t) = 2.20 + 0.028(t-0.6)^3 - 0.022t,
  %   f(t,y) = phi'(t) + (0.244+0.08t)(y-phi(t)).
  % Thus f(t,phi(t)) = phi'(t) identically. Both the curve and every
  % direction element use these same functions; f and f_y are continuous.
  \pgfmathdeclarefunction{odecurve}{1}{%
    \pgfmathparse{2.20+0.028*(#1-0.6)^3-0.022*#1}%
  }
  \pgfmathdeclarefunction{odeslope}{2}{%
    \pgfmathparse{0.084*(#1-0.6)^2-0.022+(0.244+0.08*#1)*(#2-odecurve(#1))}%
  }

  % A direction element is an UNORIENTED segment through its sample point.
  % Equal x/y scales and length normalization preserve its actual slope.
  \def\DirectionElement#1#2{%
    \pgfmathsetmacro{\fieldSlope}{odeslope(#1,#2)}%
    \pgfmathsetmacro{\halfT}{0.17/sqrt(1+\fieldSlope*\fieldSlope)}%
    \pgfmathsetmacro{\halfY}{\halfT*\fieldSlope}%
    \draw[field] ({#1-\halfT},{#2-\halfY})--({#1+\halfT},{#2+\halfY});%
    \fill (#1,#2) circle[radius=0.72pt];%
  }

  \draw[axis] (-1.85,0)--(4.90,0) node[right=2pt] {$t$};
